define function matrix_identity (matrix matrix_identity_matrix)
{
	
	param matrix_identity_matrix.m00 = 1.0
	param matrix_identity_matrix.m01 = 0.0
	param matrix_identity_matrix.m02 = 0.0

	param matrix_identity_matrix.m10 = 0.0
	param matrix_identity_matrix.m11 = 1.0
	param matrix_identity_matrix.m12 = 0.0

	param matrix_identity_matrix.m20 = 0.0
	param matrix_identity_matrix.m21 = 0.0
	param matrix_identity_matrix.m22 = 1.0

}

# cos(zero_deg) = 1
# cos(neinty_deg) = 0.0007963267107332633
# cos(one_eighty_deg) = -0.9999987317275395
# so the up tilt sensor apparently gives the y component
# so

define function build_matrix (fractional build_matrix_compass, fractional build_matrix_tilt_forward, fractional build_matrix_tilt_right, matrix build_matrix_result)
{
	define var vector build_matrix_heading
	define var vector build_matrix_up
	define var vector build_matrix_right
	define var fractional build_matrix_sr
	define var fractional build_matrix_cr
	define var fractional build_matrix_sp
	define var fractional build_matrix_cp
	
	
	param build_matrix_tilt_forward = ((param build_matrix_tilt_forward - 90.0) / 180.0) * var PI
	param build_matrix_tilt_right = ((param build_matrix_tilt_right - 90.0) / 180.0) * var PI

	var build_matrix_sr = sin (param build_matrix_tilt_right)
	var build_matrix_cr = cos (param build_matrix_tilt_right)
	var build_matrix_sp = sin (param build_matrix_tilt_forward)
	var build_matrix_cp = cos (param build_matrix_tilt_forward)
	
	param build_matrix_compass = (param build_matrix_compass / 180.0) * var PI

	var build_matrix_heading.x = sin (param build_matrix_compass) * -1.0
	var build_matrix_heading.y = 0.0
	var build_matrix_heading.z = cos (param build_matrix_compass)
	

	var build_matrix_up.x = var build_matrix_sr * -1.0
	var build_matrix_up.y = var build_matrix_cp * var build_matrix_cr
	var build_matrix_up.z = var build_matrix_sp * var build_matrix_cr
	
	call vec_cross (var build_matrix_up, var build_matrix_heading, var build_matrix_right)
	call vec_cross (var build_matrix_right, var build_matrix_up, var build_matrix_heading)
	
	call vec_normalize (var build_matrix_heading, var build_matrix_heading)
	call vec_normalize (var build_matrix_up, var build_matrix_up)
	call vec_normalize (var build_matrix_right, var build_matrix_right)
	
	param build_matrix_result.m00 = var build_matrix_right.x
	param build_matrix_result.m01 = var build_matrix_right.y
	param build_matrix_result.m02 = var build_matrix_right.z
	param build_matrix_result.m10 = var build_matrix_up.x
	param build_matrix_result.m11 = var build_matrix_up.y
	param build_matrix_result.m12 = var build_matrix_up.z
	param build_matrix_result.m20 = var build_matrix_heading.x
	param build_matrix_result.m21 = var build_matrix_heading.y
	param build_matrix_result.m22 = var build_matrix_heading.z
	
}

define function to_local (vector to_local_a, vector to_local_result)
{
	define var vector to_local_temp
	
	var to_local_temp.x = ((var matrix_world.m00 * param to_local_a.x) + (var matrix_world.m01 * param to_local_a.y)) + (var matrix_world.m02 * param to_local_a.z)
	var to_local_temp.y = ((var matrix_world.m10 * param to_local_a.x) + (var matrix_world.m11 * param to_local_a.y)) + (var matrix_world.m12 * param to_local_a.z)
	var to_local_temp.z = ((var matrix_world.m20 * param to_local_a.x) + (var matrix_world.m21 * param to_local_a.y)) + (var matrix_world.m22 * param to_local_a.z)

	call vec_copy(var to_local_temp, param to_local_result)
}

define function to_world (vector to_world_a, vector to_world_result)
{
	define var vector to_world_temp

	var to_world_temp.x = ((var matrix_world.m00 * param to_world_a.x) + (var matrix_world.m10 * param to_world_a.y)) + (var matrix_world.m20 * param to_world_a.z)
	var to_world_temp.y = ((var matrix_world.m01 * param to_world_a.x) + (var matrix_world.m11 * param to_world_a.y)) + (var matrix_world.m21 * param to_world_a.z)
	var to_world_temp.z = ((var matrix_world.m02 * param to_world_a.x) + (var matrix_world.m12 * param to_world_a.y)) + (var matrix_world.m22 * param to_world_a.z)

	call vec_copy(var to_world_temp, param to_world_result)
}
