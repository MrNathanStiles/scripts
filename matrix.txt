#function
define function matrix_identity (matrix matrix_identity_matrix)
{
	
	param matrix_identity_matrix.m00 = 1.0
	param matrix_identity_matrix.m01 = 0.0
	param matrix_identity_matrix.m02 = 0.0

	param matrix_identity_matrix.m10 = 0.0
	param matrix_identity_matrix.m11 = 1.0
	param matrix_identity_matrix.m12 = 0.0

	param matrix_identity_matrix.m20 = 0.0
	param matrix_identity_matrix.m21 = 0.0
	param matrix_identity_matrix.m22 = 1.0

}

# cos(zero_deg) = 1
# cos(neinty_deg) = 0.0007963267107332633
# cos(one_eighty_deg) = -0.9999987317275395
# so the up tilt sensor apparently gives the y component
# so

define function build_matrix (fractional build_matrix_compass, fractional build_matrix_right, fractional build_matrix_up, fractional build_matrix_forward, matrix build_matrix_result)
{
	define var fractional build_matrix_yaw
	define var fractional build_matrix_sy
	define var fractional build_matrix_cy
	
	define var fractional build_matrix_ar
	define var fractional build_matrix_au
	define var fractional build_matrix_af
	
	define var fractional build_matrix_cr
	define var fractional build_matrix_cu
	define var fractional build_matrix_cf
	define var fractional build_matrix_sf
	define var fractional build_matrix_h
	
	var build_matrix_yaw = (param build_matrix_compass / 180.0) * var PI
	var build_matrix_sy = sin (var build_matrix_yaw)
	var build_matrix_cy = cos (var build_matrix_yaw)
	
	var build_matrix_ar = (param build_matrix_right / 180.0) * var PI
	var build_matrix_au = (param build_matrix_up / 180.0) * var PI
	var build_matrix_af = (param build_matrix_forward / 180.0) * var PI
	
	var build_matrix_cr = cos (var build_matrix_ar)
	var build_matrix_cu = cos (var build_matrix_au)
	var build_matrix_cf = cos (var build_matrix_af)
	var build_matrix_sf = sin (var build_matrix_af)
	
	var build_matrix_h = var build_matrix_sf
	if (var build_matrix_h == 0.0)
	{
		var build_matrix_h = 0.0001
	}
	
	param build_matrix_result.m02 = (0.0 - var build_matrix_sy) * var build_matrix_sf
	param build_matrix_result.m12 = var build_matrix_cf
	param build_matrix_result.m22 = var build_matrix_cy * var build_matrix_sf
	
	param build_matrix_result.m00 = (var build_matrix_cy * var build_matrix_cu + var build_matrix_sy * var build_matrix_cf * var build_matrix_cr) / var build_matrix_h
	param build_matrix_result.m10 = var build_matrix_cr
	param build_matrix_result.m20 = (var build_matrix_sy * var build_matrix_cu - var build_matrix_cy * var build_matrix_cf * var build_matrix_cr) / var build_matrix_h
	
	param build_matrix_result.m01 = ((0.0 - var build_matrix_cy * var build_matrix_cr) + var build_matrix_sy * var build_matrix_cf * var build_matrix_cu) / var build_matrix_h
	param build_matrix_result.m11 = var build_matrix_cu
	param build_matrix_result.m21 = ((0.0 - var build_matrix_sy * var build_matrix_cr) - var build_matrix_cy * var build_matrix_cf * var build_matrix_cu) / var build_matrix_h
}

define function to_local (vector to_local_a, vector to_local_result)
{
	define var vector to_local_temp
	
	var to_local_temp.x = ((var matrix_world.m00 * param to_local_a.x) + (var matrix_world.m10 * param to_local_a.y)) + (var matrix_world.m20 * param to_local_a.z)
	var to_local_temp.y = ((var matrix_world.m01 * param to_local_a.x) + (var matrix_world.m11 * param to_local_a.y)) + (var matrix_world.m21 * param to_local_a.z)
	var to_local_temp.z = ((var matrix_world.m02 * param to_local_a.x) + (var matrix_world.m12 * param to_local_a.y)) + (var matrix_world.m22 * param to_local_a.z)

	call vec_copy(var to_local_temp, param to_local_result)
}

define function to_world (vector to_world_a, vector to_world_result)
{
	define var vector to_world_temp

	var to_world_temp.x = ((var matrix_world.m00 * param to_world_a.x) + (var matrix_world.m01 * param to_world_a.y)) + (var matrix_world.m02 * param to_world_a.z)
	var to_world_temp.y = ((var matrix_world.m10 * param to_world_a.x) + (var matrix_world.m11 * param to_world_a.y)) + (var matrix_world.m12 * param to_world_a.z)
	var to_world_temp.z = ((var matrix_world.m20 * param to_world_a.x) + (var matrix_world.m21 * param to_world_a.y)) + (var matrix_world.m22 * param to_world_a.z)

	call vec_copy(var to_world_temp, param to_world_result)
}

on_tick
{
	define var vector on_tick_forward
	define var vector on_tick_up
	define var vector on_tick_right
	define var vector on_tick_velo
	

	define var vector on_tick_acc
	define var fractional on_tick_len
	
	call build_matrix (out compass.north_angle, out tilt_right.tilt, out tilt_up.tilt, out tilt_forward.tilt, var matrix_world)
	
	var on_tick_forward.x = 0.0
	var on_tick_forward.y = 0.0
	var on_tick_forward.z = 1.0
	
	var on_tick_up.x = 0.0
	var on_tick_up.y = 1.0
	var on_tick_up.z = 0.0
	
	var on_tick_right.x = 1.0
	var on_tick_right.y = 0.0
	var on_tick_right.z = 0.0
	
	var on_tick_acc.x = out acc_right.acceleration
    var on_tick_acc.y = out acc_up.acceleration
    var on_tick_acc.z = out acc_forward.acceleration

	call to_world (var on_tick_acc, var on_tick_acc)
	call vec_add (var global_velo, var on_tick_acc, var global_velo)	
	call vec_scale (var global_velo, var delta, var on_tick_velo)
	call vec_add (var global_position, var on_tick_velo, var global_position)
	
	var global_position.y = out altitude.altitude
	
	in gps_x.display_value = var global_position.x
	in gps_y.display_value = var global_position.y
	in gps_z.display_value = var global_position.z

	call to_local (var on_tick_forward, var on_tick_forward)
	call to_local (var on_tick_up, var on_tick_up)
	call to_local (var on_tick_right, var on_tick_right)
	
	in disp_forward_x.display_value = var on_tick_forward.x
