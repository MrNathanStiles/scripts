define function vec_scale (fractional vec_scale_x, fractional vec_scale_y, fractional vec_scale_z, fractional vec_scale_n)
{
	param vec_scale_x = param vec_scale_x * param vec_scale_n
	param vec_scale_y = param vec_scale_y * param vec_scale_n
	param vec_scale_z = param vec_scale_z * param vec_scale_n
}

define function vec_dot (fractional vec_dot_x1, fractional vec_dot_y1, fractional vec_dot_z1, fractional vec_dot_x2, fractional vec_dot_y2, fractional vec_dot_z2, fractional vec_dot_result)
{
	param vec_dot_result = ((param vec_dot_x1 * param vec_dot_x2) + (param vec_dot_y1 * param vec_dot_y2)) + (param vec_dot_z1 * param vec_dot_z2)
}

define function vec_len_sq (fractional vec_len_sq_x, fractional vec_len_sq_y, fractional vec_len_sq_z, fractional vec_len_sq_result)
{
	call vec_dot (param vec_len_sq_x, param vec_len_sq_y, param vec_len_sq_z, param vec_len_sq_x, param vec_len_sq_y, param vec_len_sq_z, param vec_len_sq_result)
}

define function vec_len (fractional vec_len_x, fractional vec_len_y, fractional vec_len_z, fractional vec_len_result)
{
	call vec_len_sq (param vec_len_x, param vec_len_y, param vec_len_z, param vec_len_result)
	param vec_len_result = sqrt (param vec_len_result)
}

define function vec_normalize (fractional vec_normalize_x, fractional vec_normalize_y, fractional vec_normalize_z)
{
	define var fractional vec_normalize_len
	var vec_normalize_len = 0.0
	
	call vec_len (param vec_normalize_x, param vec_normalize_y, param vec_normalize_z, var vec_normalize_len)
	if (var vec_normalize_len == 0.0)
	{
		var vec_normalize_len = 1.0
	}
	var vec_normalize_len = 1.0 / var vec_normalize_len
	call vec_scale (param vec_normalize_x, param vec_normalize_y, param vec_normalize_z, var vec_normalize_len)
}

define function build_matrix (fractional build_matrix_compass, fractional build_matrix_right, fractional build_matrix_up, fractional build_matrix_forward)
{
	define var fractional build_matrix_yaw
	define var fractional build_matrix_sy
	define var fractional build_matrix_cy
	
	define var fractional build_matrix_ar
	define var fractional build_matrix_au
	define var fractional build_matrix_af
	
	define var fractional build_matrix_cr
	define var fractional build_matrix_cu
	define var fractional build_matrix_cf
	define var fractional build_matrix_sf
	define var fractional build_matrix_h
	
	var build_matrix_yaw = (param build_matrix_compass / 180.0) * var PI
	var build_matrix_sy = sin (var build_matrix_yaw)
	var build_matrix_cy = cos (var build_matrix_yaw)
	
	var build_matrix_ar = (param build_matrix_right / 180.0) * var PI
	var build_matrix_au = (param build_matrix_up / 180.0) * var PI
	var build_matrix_af = (param build_matrix_forward / 180.0) * var PI
	
	var build_matrix_cr = cos (var build_matrix_ar)
	var build_matrix_cu = cos (var build_matrix_au)
	var build_matrix_cf = cos (var build_matrix_af)
	var build_matrix_sf = sin (var build_matrix_af)
	
	var build_matrix_h = var build_matrix_sf
	if (var build_matrix_h == 0.0)
	{
		var build_matrix_h = 0.0001
	}
	
	var m02 = (0.0 - var build_matrix_sy) * var build_matrix_sf
	var m12 = var build_matrix_cf
	var m22 = var build_matrix_cy * var build_matrix_sf
	
	var m00 = (var build_matrix_cy * var build_matrix_cu + var build_matrix_sy * var build_matrix_cf * var build_matrix_cr) / var build_matrix_h
	var m10 = var build_matrix_cr
	var m20 = (var build_matrix_sy * var build_matrix_cu - var build_matrix_cy * var build_matrix_cf * var build_matrix_cr) / var build_matrix_h
	
	var m01 = ((0.0 - var build_matrix_cy * var build_matrix_cr) + var build_matrix_sy * var build_matrix_cf * var build_matrix_cu) / var build_matrix_h
	var m11 = var build_matrix_cu
	var m21 = ((0.0 - var build_matrix_sy * var build_matrix_cr) - var build_matrix_cy * var build_matrix_cf * var build_matrix_cu) / var build_matrix_h
}

define function to_local (fractional to_local_x, fractional to_local_y, fractional to_local_z)
{
	define var fractional to_local_tx
	define var fractional to_local_ty
	define var fractional to_local_tz
	
	var to_local_tx = ((var m00 * param to_local_x) + (var m10 * param to_local_y)) + (var m20 * param to_local_z)
	var to_local_ty = ((var m01 * param to_local_x) + (var m11 * param to_local_y)) + (var m21 * param to_local_z)
	var to_local_tz = ((var m02 * param to_local_x) + (var m12 * param to_local_y)) + (var m22 * param to_local_z)
	
	param to_local_x = var to_local_tx
	param to_local_y = var to_local_ty
	param to_local_z = var to_local_tz
}

define function to_world (fractional to_world_x, fractional to_world_y, fractional to_world_z)
{
	define var fractional to_world_tx
	define var fractional to_world_ty
	define var fractional to_world_tz
	
	var to_world_tx = ((var m00 * param to_world_x) + (var m01 * param to_world_y)) + (var m02 * param to_world_z)
	var to_world_ty = ((var m10 * param to_world_x) + (var m11 * param to_world_y)) + (var m12 * param to_world_z)
	var to_world_tz = ((var m20 * param to_world_x) + (var m21 * param to_world_y)) + (var m22 * param to_world_z)
	
	param to_world_x = var to_world_tx
	param to_world_y = var to_world_ty
	param to_world_z = var to_world_tz
}
