define var fractional PI

define function deg_to_rad(fractional deg_to_rad_value)
{
    param deg_to_rad_value = param deg_to_rad_value * (var PI / 180.0)
}

define function clamp (fractional clamp_value, fractional clamp_min, fractional clamp_max)
{
	if (param clamp_value < param clamp_min)
	{
		param clamp_value = param clamp_min
	}
	elseif (param clamp_value > param clamp_max)
	{
		param clamp_value = param clamp_max
	}
}
