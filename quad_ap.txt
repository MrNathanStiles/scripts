define type pid_data
{
	define member fractional Kp
	define member fractional Ki
	define member fractional Kd
	define member fractional i
	define member fractional le
}

define var integer tick
define var fractional altitude
define var fractional target_altitude
define var fractional delta
define var fractional temp_collective
define var pid_data pid_alt
define var pid_data pid_pitch
define var pid_data pid_roll

define function clamp (fractional clamp_value, fractional clamp_min, fractional clamp_max)
{
	if (param clamp_value < param clamp_min)
	{
		param clamp_value = param clamp_min
	}
	elseif (param clamp_value > param clamp_max)
	{
		param clamp_value = param clamp_max
	}
}
define function make_pid (pid_data make_pid_data, fractional make_pid_p, fractional make_pid_i, fractional make_pid_d)
{
	param make_pid_data.Kp = param make_pid_p
	param make_pid_data.Ki = param make_pid_i
	param make_pid_data.Kd = param make_pid_d
	param make_pid_data.i = 0.0
	param make_pid_data.le = 0.0
}

define function run_pid (pid_data run_pid_data, fractional run_pid_sp, fractional run_pid_pv, fractional run_pid_result)
{
	define var fractional run_pid_e
	define var fractional run_pid_d
	
	var run_pid_e = param run_pid_sp - param run_pid_pv
	param run_pid_data.i = param run_pid_data.i + var run_pid_e * param run_pid_data.Ki
	var run_pid_d = var run_pid_e - param run_pid_data.le
	param run_pid_data.le = var run_pid_e
	param run_pid_result = var run_pid_e * param run_pid_data.Kp + param run_pid_data.i + var run_pid_d * param run_pid_data.Kd
}

define function sign (fractional sign_value, fractional sign_result)
{
	if (param sign_value < 0.0)
	{
		param sign_result = -1.0
	}
	else
	{
		param sign_result = 1.0
	}
}

var temp_collective = 0.3
var tick = 0
var altitude = 0.0
var target_altitude = 0.0
var delta = 1.0 / 60.0

on_tick
{
	define var fractional roll
	define var fractional pitch
	define var fractional collective
	define var fractional vertical_velocity
	define var fractional vertical_velocity_error
	define var fractional altitude_error
	define var fractional altitude_error_direction
	define var fractional target_vertical_velocity
	define var fractional acceleration
	
	var acceleration = 0.1
	
	var tick = var tick + 1
	var vertical_velocity = (out altitude.altitude - var altitude) * 60.0
	var altitude = out altitude.altitude

	in disp_altitude.display_value = var altitude
	in disp_target_altitude.display_value = var target_altitude
	
	
	
	if (out system_on.is_pressed)
	{
		var altitude_error = var target_altitude - out altitude.altitude
		call sign (var altitude_error, var altitude_error_direction)
		var target_vertical_velocity = sqrt (abs ((2.0 * var acceleration) * var altitude_error)) * var altitude_error_direction
		
		if (var target_vertical_velocity > 1.0)
		{
			var target_vertical_velocity = 1.0
		}
		elseif (var target_vertical_velocity < -1.0)
		{
			var target_vertical_velocity = -1.0
		}

		if (abs (var altitude_error) < 1.0)
		{
			var target_vertical_velocity = var target_vertical_velocity * 0.1
		}

		call run_pid (var pid_alt, var target_vertical_velocity, var vertical_velocity, var collective)
		var collective = var collective + 0.1 
		
		

		#pitch going into the pid at + comes out as - when setpoint is 0
		var pitch = (90.0 - out tilt_forward.tilt) / 90.0
		call run_pid (var pid_pitch, out stick.pitch * -0.3, var pitch, var pitch)
		
		#roll going into the pid at + comes out as - when setpoint is 0
		var roll = (90.0 - out tilt_right.tilt) / 90.0
		call run_pid (var pid_roll, out stick.roll * -0.3, var roll, var roll)

		in disp_collective.display_value = var collective
		in disp_pitch.display_value = var pitch
		in disp_roll.display_value = var roll
		in rotor_fl.input = (var collective - var roll) + var pitch
		in rotor_fr.input = (var collective + var roll) + var pitch
		
		in rotor_bl.input = (var collective - var roll) - var pitch
		in rotor_br.input = (var collective + var roll) - var pitch
		
		var target_altitude = var target_altitude + (out stick.right_pedal * 0.1)
		var target_altitude = var target_altitude - (out stick.left_pedal * 0.1)
	}
	else
	{
		call make_pid(var pid_alt, 0.3, 0.003, 0.0)
		call make_pid(var pid_pitch, 0.9, 0.005, 0.0)
		call make_pid(var pid_roll, 0.9, 0.005, 0.0)
		in rotor_fl.input = 0.0
		in rotor_fr.input = 0.0
		in rotor_bl.input = 0.0
		in rotor_br.input = 0.0
		var target_altitude = var altitude + 2.0
	}
	
}
