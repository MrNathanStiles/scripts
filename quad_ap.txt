define var fractional collective
define var integer tick
define var fractional altitude
define var fractional target_altitude

var collective = 0.0
var tick = 0
var altitude = 0.0
var target_altitude = 0.0

#x = gn(4)
#y = gn(5)
#z = gn(6)

cx, sx = cos(x), sin(x)
cy, sy = cos(y), sin(y)
cz, sz = cos(z), sin(z)

on_tick
{
	define var fractional on_tick_roll
	define var fractional on_tick_pitch
	
	var tick = var tick + 1
	
	var altitude = out altitude.altitude
	
	
	
	
	
	var collective = var collective + (out stick.right_pedal * 0.001)
	var collective = var collective - (out stick.left_pedal * 0.001)
	
	if (var collective > 0.5)
	{
		var collective = 0.5
	}
	
	if (var collective < -0.5)
	{
		var collective = -0.5
	}
	
	var on_tick_roll = ((90.0 - out tilt_right.tilt) / 90.0) + out stick.roll
	var on_tick_pitch = (90.0 - out tilt_forward.tilt) / 90.0
	
	#in dial_forward.is_illuminated = true
	#in dial_right.is_illuminated = true
	#in dial_forward.display_value = var pitch
	#in dial_right.display_value = var roll
	
	in rotor_fl.input = var collective - var on_tick_roll + var on_tick_pitch
	in rotor_fr.input = var collective + var on_tick_roll + var on_tick_pitch
	
	in rotor_bl.input = var collective - var on_tick_roll - var on_tick_pitch
	in rotor_br.input = var collective + var on_tick_roll - var on_tick_pitch
	
	
}
