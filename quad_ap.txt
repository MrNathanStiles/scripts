define var fractional collective
define var integer tick

define var fractional target_altitude

var collective = 0.0
var tick = -1
var altitude = 0
var target_altitude = 0.0

#x = gn(4)

#z = gn(6)

cx, sx = cos(x), sin(x)
cy, sy = cos(y), sin(y)
cz, sz = cos(z), sin(z)

on_tick
{
	var tick = var tick + 1
	
	var  = out altitude.altutude - var altitude
	var altitude = out altitude.altutude
	
	
	
	
	define var fractional roll
	define var fractional pitch
	
	var collective = var collective + (out stick.right_pedal * 0.001)
	var collective = var collective - (out stick.left_pedal * 0.001)
	
	if (var collective > 0.5)
	{
		var collective = 0.5
	}
	
	if (var collective < -0.5)
	{
		var collective = -0.5
	}
	
	var pitch = (90.0 - out tilt_forward.tilt) / 90.0
	var roll = ((90.0 - out tilt_right.tilt) / 90.0) + out stick.roll
	
	in dial_forward.is_illuminated = true
	in dial_right.is_illuminated = true
	in dial_forward.display_value = var pitch
	in dial_right.display_value = var roll
	
	in FL.input = var collective - var roll
	in FR.input = var collective + var roll
	
	in BL.input = var collective - var roll
	in BR.input = var collective + var roll
	
	
}
