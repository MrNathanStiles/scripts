# trapdoor on stormworks discord
# x = target
# y = current
# ((x-y)%1+1.5)%1-0.5
define function wrap (fractional wrap_target, fractional wrap_current, fractional wrap_result)
{
	param wrap_result = param wrap_target - param wrap_current
	call fmod (param wrap_result, 1.0, param wrap_result)
	param wrap_result = param wrap_result + 1.5
	call fmod (param wrap_result, 1.0, param wrap_result)
	param wrap_result = param wrap_result - 0.5
}

define var fractional motor_x_0_position
define var fractional motor_x_1_position
define var fractional motor_y_0_position

# the radius in rps
define var fractional radius
define var fractional target_angle

define var integer tick
define var integer state_calibrate_estimate
define var integer state_calibrate_right
define var integer state_calibrate_center
define var integer state_run
define var integer state
define var integer state_start

var tick = 0
var state_calibrate_estimate = 0
var state_calibrate_right = 1
var state_calibrate_center = 2
var state_run = 3
var state = var state_calibrate_estimate
var state_start = 0

var motor_x_0_position = 0.0
var motor_x_1_position = 0.0
var motor_y_0_position = 0.0


var target_angle = 0.0

# compass - is looking east of north
# compass + is looking west of north
# compass / 180.0

# 12.6mm per revolution
# 8cm per voxel
# radius (16(voxels) x 8(cm) * 10(mm per cm)) / 12.6(mm per revolution)
var radius = ((16.0 * 8.0) * 10.0) / 12.6


on_tick
{
	var tick = var tick + 1
	
	define var fractional base_north
	define var fractional arm_north
	define var fractional arm_error
	define var fractional current_angle
	define var fractional motor_stopped_speed
	define var fractional temp_target_angle
	define var fractional x
	define var fractional y
	
	define var integer state_age
	
	var motor_stopped_speed = 0.1
	var state_age = var tick - var state_start
	
	define var bool motors_stopped
	
	var motors_stopped = false
	
	if (((abs (out motor_x_0.rps) < var motor_stopped_speed) && (abs (out motor_x_1.rps) < var motor_stopped_speed)) && (abs (out motor_y_0.rps) < var motor_stopped_speed))
	{
		var motors_stopped = true
	}

	call atan2 (var motor_y_0_position, var motor_x_0_position, var current_angle)
	
	
	
	
	var motor_x_0_position = var motor_x_0_position + out motor_x_0.rps * var delta
	var motor_x_1_position = var motor_x_1_position + out motor_x_1.rps * var delta
	var motor_y_0_position = var motor_y_0_position + out motor_y_0.rps * var delta
	
	#left of north is positive
	var base_north = out compass_base.north_angle / 360.0
	var arm_north = out compass_arm.north_angle / 360.0
	call wrap (var base_north, var arm_north, var arm_error)
	
	# arm_error - is left(above in atan2 view) of base
	# arm_error + is right(below in atan2 view) of base
	# turret center to base compass direction is x = +1, y = 0
	
	
	in disp_arm_error.display_value = var arm_error
	
	if (var state == var state_run)
	{
		
	}
	elseif (var state == var state_calibrate_estimate)
	{
		var motor_x_0_position = var radius * cos (var arm_error * var PI)
		var motor_x_1_position = var motor_x_0_position
		var motor_y_0_position = var radius * sin (var arm_error * var PI)
		
		var target_angle = 0.0
		
		in motor_x_0.target_rps = out motor_x_0.rps * -1.0
		in motor_x_1.target_rps = out motor_x_1.rps * -1.0
		in motor_y_0.target_rps = out motor_y_0.rps * -1.0
		if (var motors_stopped)
		{
			if (var state_age > 60)
			{
				var state = var state_calibrate_right
				var state_start = var tick
			}
		}
		else
		{
			var state_start = var tick
		}
	}
	elseif (var state == var state_calibrate_right)
	{
		#var state = var state_calibrate_center
		#var state_start = var tick

		if (abs (var arm_error) < var PI / 2.0)
		{
			var state = var state_calibrate_center
			var state_start = var tick
		}
		else
		{
			call lerp(var current_angle, var target_angle, 0.1, var temp_target_angle)
			var x = var radius * cos (var temp_target_angle)
			var y = var radius * sin (var temp_target_angle)
			in motor_x_0.target_rps = (var x - var motor_x_0_position) * -1.0
			in motor_x_1.target_rps = (var x - var motor_x_1_position) * -1.0
			in motor_y_0.target_rps = (var y - var motor_y_0_position) * 1.0
		}


		
	}
	elseif (var state == var state_calibrate_center)
	{
		call lerp(var current_angle, var target_angle, 0.1, var temp_target_angle)
		var x = var radius * cos (var temp_target_angle)
		var y = var radius * sin (var temp_target_angle)
		in motor_x_0.target_rps = (var x - var motor_x_0_position) * -11.0
		in motor_x_1.target_rps = (var x - var motor_x_1_position) * -11.0
		in motor_y_0.target_rps = (var y - var motor_y_0_position) * 11.0
	}
	
	
	
	
	in disp_state.display_value = to_fractional (var state)

	# x motors - moves along + axis
	# y motors + moves along + axis
	
	#in motor_x_0.target_rps = -1.0
	#in motor_x_1.target_rps = -1.0
	#in motor_y_0.target_rps = 1.0

	
	
	
}
