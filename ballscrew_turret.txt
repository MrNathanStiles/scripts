define var fractional PI

define type pid_data
{
	define member fractional Kp
	define member fractional Ki
	define member fractional Kd
	define member fractional i
	define member fractional le
}

var PI = 3.141592

define function atan2(fractional atan2_y, fractional atan2_x, fractional atan2_result)
{
	if (param atan2_x > 0.0)
	{
        param atan2_result = atan (param atan2_y / param atan2_x)
    }
    elseif ((param atan2_x < 0.0) && (param atan2_y >= 0.0))
	{
        param atan2_result = atan (param atan2_y / param atan2_x) + var PI
    }
    elseif ((param atan2_x < 0.0) && (param atan2_y < 0.0))
	{
        param atan2_result = atan (param atan2_y / param atan2_x) - var PI
    }
    elseif ((param atan2_x == 0.0) && (param atan2_y > 0.0))
	{
        param atan2_result = var PI / 2.0
    }
    elseif ((param atan2_x == 0.0) && (param atan2_y < 0.0))
	{
        param atan2_result = 0.0 - (var PI / 2.0)
    }
    else
	{
        param atan2_result = 0.0
    }
}

define function make_pid (pid_data make_pid_data, fractional make_pid_p, fractional make_pid_i, fractional make_pid_d)
{
	param make_pid_data.Kp = param make_pid_p
	param make_pid_data.Ki = param make_pid_i
	param make_pid_data.Kd = param make_pid_d
	param make_pid_data.i = 0.0
	param make_pid_data.le = 0.0
}

define function run_pid (pid_data run_pid_data, fractional run_pid_sp, fractional run_pid_pv, fractional run_pid_result)
{
	define var fractional run_pid_e
	define var fractional run_pid_d
	
	var run_pid_e = param run_pid_sp - param run_pid_pv
	param run_pid_data.i = param run_pid_data.i + var run_pid_e * param run_pid_data.Ki
	var run_pid_d = var run_pid_e - param run_pid_data.le
	param run_pid_data.le = var run_pid_e
	param run_pid_result = var run_pid_e * param run_pid_data.Kp + param run_pid_data.i + var run_pid_d * param run_pid_data.Kd
}

define function fmod (fractional fmod_a, fractional fmod_b, fractional fmod_result)
{
	define var integer fmod_ta
	define var integer fmod_tb
	
	var fmod_ta = to_integer (param fmod_a * 1000000.0)
	var fmod_tb = to_integer (param fmod_b * 1000000.0)
	param fmod_result = to_fractional (var fmod_ta % var fmod_tb) / 1000000.0
}

# trapdoor on stormworks discord
# x = target
# y = current
# ((x-y)%1+1.5)%1-0.5
define function wrap (fractional wrap_target, fractional wrap_current, fractional wrap_result)
{
	param wrap_result = param wrap_target - param wrap_current
	call fmod(param wrap_result, 1.0, param wrap_result)
	param wrap_result = param wrap_result + 1.5
	call fmod(param wrap_result, 1.0, param wrap_result)
	param wrap_result = param wrap_result - 0.5
}
