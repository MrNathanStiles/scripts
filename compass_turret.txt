define var fractional target
define var fractional Kp
define var fractional Ki
define var fractional Kd
define var fractional i
define var fractional le

var target = 0.0

var Kp = 7.0
var Ki = 0.0
var Kd = 599.0

var i = 0.0
var le = 0.0

define function run (fractional run_sp, fractional run_pv, fractional run_result)
{
	define var fractional run_e
	define var fractional run_d

	var run_e = param run_sp - param run_pv
	var i = var i + var run_e * var Ki
	var run_d = (var run_e - var le)
	var le = var run_e
	param run_result =  var Kp * var run_e + var i + var Kd * var run_d
}

define function fmod (fractional fmod_a, fractional fmod_b, fractional fmod_result)
{
	define var integer fmod_ta
	define var integer fmod_tb
	
	var fmod_ta = to_integer (param fmod_a * 1000000.0)
	var fmod_tb = to_integer (param fmod_b * 1000000.0)
	param fmod_result = to_fractional (var fmod_ta % var fmod_tb) / 1000000.0
}

on_tick
{
	define var fractional error
	
	var target = var target + out stick.roll * -1.0
	
	
	var error = (var target - out compass.north_angle)
	var error = var error / 360.0
	call fmod (var error, 1.0, var error)
	var error = var error + 1.5
	call fmod (var error, 1.0, var error)
	var error = var error - 0.5

	var error = var error * -3.0
	
	call run (var error, 0.0, var error)

	# x = target
	# y = current
	# ((x-y)%1+1.5)%1-0.5


	
	in m1.target_rps = var error
	in m2.target_rps = var error
	in m3.target_rps = var error
	in m4.target_rps = var error
	in disp_heading.display_value = abs (out compass.north_angle)
	in disp_target.display_value = abs (var target)
	
}
