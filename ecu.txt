#use math
#use pid
#use clamp

define var pid_data pid_throttle
define var fractional base_throttle
define var fractional PI
define var integer tick

define var integer state_off
define var integer state_start
define var integer state_run
define var integer state
define var fractional target_throttle

define var fractional target_rps

var target_rps = 90.0
var state_off = 0
var state_start = 1
var state_run = 2
var state = var state_off
var base_throttle = 0.0

var tick = 0


define function init ()
{
	call make_pid (var pid_throttle, 1.0 / (var target_rps - 10.0), 0.001, 0.0)
	var target_throttle = 0.0
}

on_tick
{
	define var fractional rps
	define var bool start
	
	var rps = out engine.rps
	var tick = var tick + 1
	
	if (var tick == 1)
	{
		call init ()
	}
	
	if (out run.is_pressed)
	{
		if (var state == var state_off)
		{
			if (out engine.rps > 7.0)
			{
				var state = var state_run
			}
			else
			{
				var state = var state_start
			}
		}
		elseif (var state == var state_start)
		{
			var start = true
			var target_throttle = 1.0
			if (out engine.rps > 7.0)
			{
				var state = var state_run
			}
		}
		elseif (var state == var state_run)
		{
			if (var rps < 6.0)
			{
				var state = var state_off
			}
			else
			{
				call run_pid (var pid_throttle, var target_rps, var rps, var target_throttle)
			}
		}
	}
	else
	{
		call init ()
	}
	
	if (var start)
	{
		in starter.target_rps = 30.0
	}
	else
	{
		in starter.target_rps = 0.0
	}
	call clamp (var target_throttle, 0.0, 1.0)
	in throttle.input = var target_throttle
}
