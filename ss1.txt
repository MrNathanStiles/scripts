define var fractional pi
define var fractional wheelcircumference
define var fractional mps
define var fractional throttle_set
define var fractional engineRps
define var fractional throttleInput
define var fractional handbrakevalue
define var fractional speedfactor
define var fractional zabs
define var bool engon
on_tick
{
	var pi = 3.1415
	var wheelcircumference = var pi * 0.3
	var mps = var wheelcircumference * abs (out Wheel.rps)
	in rps.display_value = var engineRps / 10.0
	if ((out SW.right_pedal >= 0.15) && (out ignition.value >= 0.1))
	{
		var engon = true
	}
	else
	{
		var engon = false
	}
	in gear.display_value = to_fractional (out Gearbox.current_gear)
	if (out ignition.value >= 0.5)
	{
		in airm.input = out SW.right_pedal + (out ignition.value * 0.3)
	}
	if ((var engon == true) && (out Engine.rps <= 10.0))
	{
		in starter.input = 1.0
	}
	else
	{
		in starter.input = 0.0
	}
	if ((var engon == false) || (out handbrake.is_on == true))
	{
		in airm.input = 0.0
	}
	if (out ignition.value >= 0.5)
	{
		in master.input = 1.0
	}
	else
	{
		in master.input = 0.0
	}
	if (out handbrake.is_on == true)
	{
		var handbrakevalue = 1.0
	}
	else
	{
		var handbrakevalue = 0.0
	}
	var speedfactor = 1.0 - (var mps / 20.0)
	var zabs = out SW.left_pedal * var speedfactor
	in brakes.input = var handbrakevalue + var zabs
	if (out Gearbox.current_gear == -1)
	{
		in reverse.is_illuminated = true
	}
	else
	{
		in reverse.is_illuminated = false
	}
	in steer.input = out SW.steering * var speedfactor
}
