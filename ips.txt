


define var fractional PI
define var fractional TAU

define var fractional m00
define var fractional m01
define var fractional m02

define var fractional m10
define var fractional m11
define var fractional m12

define var fractional m20
define var fractional m21
define var fractional m22

var PI = 3.141592

var m00 = 1.0
var m01 = 0.0
var m02 = 0.0

var m10 = 0.0
var m11 = 1.0
var m12 = 0.0

var m20 = 0.0
var m21 = 0.0
var m22 = 1.0

define function vec_scale (fractional vec_scale_x, fractional vec_scale_y, fractional vec_scale_z, fractional vec_scale_n)
{
	param vec_scale_x = param vec_scale_x * param vec_scale_n
	param vec_scale_y = param vec_scale_y * param vec_scale_n
	param vec_scale_z = param vec_scale_z * param vec_scale_n
}


define function vec_dot (fractional vec_dot_x1, fractional vec_dot_y1, fractional vec_dot_z1, fractional vec_dot_x2, fractional vec_dot_y2, fractional vec_dot_z2, fractional vec_dot_result)
{
	param vec_dot_result = ((param vec_dot_x1 * param vec_dot_x2) + (param vec_dot_y1 * param vec_dot_y2)) + (param vec_dot_z1 * param vec_dot_z2)
}

define function vec_len_sq (fractional vec_len_sq_x, fractional vec_len_sq_y, fractional vec_len_sq_z, fractional vec_len_sq_result)
{
	call vec_dot (param vec_len_sq_x, param vec_len_sq_y, param vec_len_sq_z, param vec_len_sq_x, param vec_len_sq_y, param vec_len_sq_z, param vec_len_sq_result)
}

define function vec_len (fractional vec_len_x, fractional vec_len_y, fractional vec_len_z, fractional vec_len_result)
{
	call vec_len_sq (param vec_len_x, param vec_len_y, param vec_len_z, param vec_len_result)
	param vec_len_result = sqrt (param vec_len_result)
}

define function vec_normalize (fractional vec_normalize_x, fractional vec_normalize_y, fractional vec_normalize_z)
{
	define var fractional vec_normalize_len
	var vec_normalize_len = 0.0
	
	call vec_len (param vec_normalize_x, param vec_normalize_y, param vec_normalize_z, var vec_normalize_len)
	if (var vec_normalize_len == 0.0)
	{
		var vec_normalize_len = 1.0
	}
	var vec_normalize_len = 1.0 / var vec_normalize_len
	call vec_scale (param vec_normalize_x, param vec_normalize_y, param vec_normalize_z, var vec_normalize_len)
}

define function build_matrix (fractional build_matrix_compass, fractional build_matrix_right, fractional build_matrix_up, fractional build_matrix_forward)
{
	
	define var fractional build_matrix_yaw
	define var fractional build_matrix_ux
	define var fractional build_matrix_uy
	define var fractional build_matrix_uz
	define var fractional build_matrix_ulen
	define var fractional build_matrix_sy
	define var fractional build_matrix_cy
	define var fractional build_matrix_nx
	define var fractional build_matrix_ny
	define var fractional build_matrix_nz
	define var fractional build_matrix_rx
	define var fractional build_matrix_ry
	define var fractional build_matrix_rz
	define var fractional build_matrix_rlen
	define var fractional build_matrix_fx
	define var fractional build_matrix_fy
	define var fractional build_matrix_fz
	
	var build_matrix_ux = (90.0 - param build_matrix_right) / 90.0
	var build_matrix_uy = (90.0 - param build_matrix_up) / 90.0
	var build_matrix_uz = (90.0 - param build_matrix_forward) / 90.0
	
	call vec_len (var build_matrix_ux, var build_matrix_uy, var build_matrix_uz, var build_matrix_ulen)
	if (var build_matrix_ulen > 0.0)
	{
		var build_matrix_ux = var build_matrix_ux / var build_matrix_ulen
		var build_matrix_uy = var build_matrix_uy / var build_matrix_ulen
		var build_matrix_uz = var build_matrix_uz / var build_matrix_ulen
	}
	
	var build_matrix_yaw = (param build_matrix_compass / 180.0) * var PI
	var build_matrix_sy = sin (var build_matrix_yaw)
	var build_matrix_cy = cos (var build_matrix_yaw)
	
	var build_matrix_nx = var build_matrix_sy
	var build_matrix_ny = 0.0
	var build_matrix_nz = var build_matrix_cy
	
	var build_matrix_rx = var build_matrix_cy
	var build_matrix_ry = 0.0
	var build_matrix_rz = 0.0 - var build_matrix_sy
	
	var build_matrix_fx = (var build_matrix_uy * var build_matrix_rz) - (var build_matrix_uz * var build_matrix_ry)
	var build_matrix_fy = (var build_matrix_uz * var build_matrix_rx) - (var build_matrix_ux * var build_matrix_rz)
	var build_matrix_fz = (var build_matrix_ux * var build_matrix_ry) - (var build_matrix_uy * var build_matrix_rx)
	
	call vec_len (var build_matrix_fx, var build_matrix_fy, var build_matrix_fz, var build_matrix_rlen)
	if (var build_matrix_rlen > 0.0)
	{
		var build_matrix_fx = var build_matrix_fx / var build_matrix_rlen
		var build_matrix_fy = var build_matrix_fy / var build_matrix_rlen
		var build_matrix_fz = var build_matrix_fz / var build_matrix_rlen
	}
	
	var build_matrix_rx = (var build_matrix_fy * var build_matrix_uz) - (var build_matrix_fz * var build_matrix_uy)
	var build_matrix_ry = (var build_matrix_fz * var build_matrix_ux) - (var build_matrix_fx * var build_matrix_uz)
	var build_matrix_rz = (var build_matrix_fx * var build_matrix_uy) - (var build_matrix_fy * var build_matrix_ux)
	
	var m00 = var build_matrix_rx
	var m01 = var build_matrix_ry
	var m02 = var build_matrix_rz
	var m10 = var build_matrix_ux
	var m11 = var build_matrix_uy
	var m12 = var build_matrix_uz
	var m20 = var build_matrix_fx
	var m21 = var build_matrix_fy
	var m22 = var build_matrix_fz
}

define function to_local (fractional to_local_x, fractional to_local_y, fractional to_local_z)
{
	define var fractional to_local_tx
	define var fractional to_local_ty
	define var fractional to_local_tz
	
	var to_local_tx = ((var m00 * param to_local_x) + (var m01 * param to_local_y)) + (var m02 * param to_local_z)
	var to_local_ty = ((var m10 * param to_local_x) + (var m11 * param to_local_y)) + (var m12 * param to_local_z)
	var to_local_tz = ((var m20 * param to_local_x) + (var m21 * param to_local_y)) + (var m22 * param to_local_z)
	
	param to_local_x = var to_local_tx
	param to_local_y = var to_local_ty
	param to_local_z = var to_local_tz
}

define function to_world (fractional to_world_x, fractional to_world_y, fractional to_world_z)
{
	define var fractional to_world_tx
	define var fractional to_world_ty
	define var fractional to_world_tz
	
	var to_world_tx = ((var m00 * param to_world_x) + (var m10 * param to_world_y)) + (var m20 * param to_world_z)
	var to_world_ty = ((var m01 * param to_world_x) + (var m11 * param to_world_y)) + (var m21 * param to_world_z)
	var to_world_tz = ((var m02 * param to_world_x) + (var m12 * param to_world_y)) + (var m22 * param to_world_z)
	
	param to_world_x = var to_world_tx
	param to_world_y = var to_world_ty
	param to_world_z = var to_world_tz
}

on_tick
{
	define var fractional on_tick_forward_x
	define var fractional on_tick_forward_y
	define var fractional on_tick_forward_z
	
	define var fractional on_tick_up_x
	define var fractional on_tick_up_y
	define var fractional on_tick_up_z
	
	define var fractional on_tick_right_x
	define var fractional on_tick_right_y
	define var fractional on_tick_right_z
	
	define var fractional on_tick_len
	
	var on_tick_forward_x = 0.0
	var on_tick_forward_y = 0.0
	var on_tick_forward_z = 1.0
	
	var on_tick_up_x = 0.0
	var on_tick_up_y = 1.0
	var on_tick_up_z = 0.0
	
	var on_tick_right_x = 1.0
	var on_tick_right_y = 0.0
	var on_tick_right_z = 0.0
	
	call build_matrix (out compass.north_angle, out tilt_right.tilt, out tilt_up.tilt, out tilt_forward.tilt)
	call to_local (var on_tick_forward_x, var on_tick_forward_y, var on_tick_forward_z)
	call to_local (var on_tick_up_x, var on_tick_up_y, var on_tick_up_z)
	call to_local (var on_tick_right_x, var on_tick_right_y, var on_tick_right_z)
	
	in disp_forward_x.display_value = var on_tick_forward_x
	in disp_forward_y.display_value = var on_tick_forward_y

	in disp_up_x.display_value = var on_tick_up_x
	in disp_up_y.display_value = var on_tick_up_y
	in disp_up_z.display_value = var on_tick_up_z
	
	in disp_right_x.display_value = var on_tick_right_x
	in disp_right_y.display_value = var on_tick_right_y
	in disp_right_z.display_value = var on_tick_right_z
	
	call vec_len (var on_tick_forward_x, var on_tick_forward_y, var on_tick_forward_z, var on_tick_len)
	in disp_forward_len.display_value = var on_tick_len
	
	call vec_len (var on_tick_up_x, var on_tick_up_y, var on_tick_up_z, var on_tick_len)
	in disp_up_len.display_value = var on_tick_len
	
	call vec_len (var on_tick_right_x, var on_tick_right_y, var on_tick_right_z, var on_tick_len)
	in disp_right_len.display_value = var on_tick_len
	
	
	
	
}
